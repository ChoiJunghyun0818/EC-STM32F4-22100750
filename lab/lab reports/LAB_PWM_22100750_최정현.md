## Problem 1: RC servo motor

An RC servo motor is a lightweight motor with high output power. It is used to control rotation angles of approximately 180 degrees (90 degrees in each direction) and is commonly applied in RC cars and small-scale robots. The angle of the motor can be controlled by the pulse width (duty ratio) of a PWM signal. The PWM period should be set to 20ms (or 50Hz). Refer to the datasheet of the RC servo motor for detailed specifications

```c
#include "ecRCC2.h"
#include "ecGPIO2.h"
#include "ecPWM2.h"
#include "ecTIM2.h"
#include "ecEXTI2.h"

#define PWM_PIN     PA_1
#define BUTTON_PIN  PC_13

volatile int angle = 0;
volatile int direction = 1;

void setup(void);
void TIM3_IRQHandler(void);
void EXTI15_10_IRQHandler(void);

int main(void) {
    setup();

    while (1) {
        // All control logic handled by interrupts
    }

void setup(void) {

    // System clock: 84 MHz
    RCC_PLL_init();

    // Servo mototr PWM: TIM2_CH2, 20ms period
    PWM_init(PA_1, 20);
    PWM_pulsewidth_us(PWM_PIN, 500);

    // TIM3 interrupt: 500ms
    TIM_UI_init(TIM3, 500);
    TIM_UI_enable(TIM3);

    NVIC_SetPriority(TIM3_IRQn, 1);
    NVIC_EnableIRQ(TIM3_IRQn);

    // button PC_13 setting
    GPIO_init(BUTTON_PIN, INPUT);
    GPIO_pupd(BUTTON_PIN, EC_PU);

    EXTI_init(BUTTON_PIN, FALL, 0);

    NVIC_SetPriority(EXTI15_10_IRQn, 0);
    NVIC_EnableIRQ(EXTI15_10_IRQn);
}


// TIM3: Every 500ms
void TIM3_IRQHandler(void) {

    if (is_UIF(TIM3)) {

        clear_UIF(TIM3);

        // angle change by 10 degrees
        angle += direction * 10;

        // Reverse direction at boundaries
        if (angle >= 180) {
            angle = 180;
            direction = -1;
        }

        else if (angle <= 0) {
            angle = 0;
            direction = 1;
        }

        // Convert angle to pulse width(500~2500us)
        uint32_t pulse_width = 500 + (angle * 2000) / 180;

        PWM_pulsewidth_us(PWM_PIN, pulse_width);
    }
}


// EXTI: Button pressed
void EXTI15_10_IRQHandler(void) {

    if (is_pending_EXTI(BUTTON_PIN)) {

        // Clear interrupt flag
        clear_pending_EXTI(BUTTON_PIN);

        // Reset motor state
        angle = 0;
        direction = 1;

        // Move motor to 0 degrees
        PWM_pulsewidth_us(PWM_PIN, 500);
    }
}
```

#### Discussion

1. Derive a simple logic to calculate CRR and ARR values to generate x\[Hz] and y\[%] duty ratio of PWM. How can you read the values of input clock frequency and PSC?

> To derive the ARR and CCR for a PWM signal with frequency and duty ratio, start by calculating the PWM period $T_{\text{PWM}} = \frac{1}{x}$. The timer's counting clock frequency depends on the timer input clock ($f_{\text{CLK}}$) divided by the prescaler plus one ($f_{\text{timer}} = \frac{f_{\text{CLK}}}{\text{PSC} + 1}$), giving a single tick duration of $T_{\text{tick}} = \frac{\text{PSC} + 1}{f_{\text{CLK}}}$. The total number of timer counts required for one full PWM period determines the ARR value, calculated as $\text{ARR} = \left(\frac{f_{\text{CLK}}}{x \cdot (\text{PSC} + 1)}\right) - 1$. Once ARR is set, the pulse high-duration counter value CCR is derived directly from the percentage duty ratio using $\text{CCR} = (\text{ARR} + 1) \cdot \frac{y}{100}$. In STM32 bare-metal programming, is is possible to read the input clock frequency and prescaler dynamically from system and hardware registers. the timer prescaler value is read directly from the timer's prescaler register (TIMx->PSC), while the timer clock frequency $f_{\text{CLK}}$ is obtained either via CMSIS RCC system clock functions like SystemCoreClock (scaled by APB prescalers from RCC registers) or by retrieving the configured clock rate from your course library initialization header (ecRCC2.h).

2. What is the smallest and highest PWM frequency that can be generated for Q1?

> For TIM2 operating on an 84 MHz timer clock on the STM32F4 series, the maximum and minimum achievable PWM frequencies are based on the limits of its 16-bit Prescaler (PSC) and Auto-Reload Register (ARR). The lowest possible PWM frequency is achieved by maximizing both registers to their 16-bit capacity ($\text{PSC} = 65535$ and $\text{ARR} = 65535$), which yields $f_{\text{PWM, min}} = \frac{84,000,000}{(65535 + 1) \cdot (65535 + 1)} \approx 0.01956\text{ Hz}$ (a period of approximately $51.12\text{ seconds}$). The highest theoretical PWM frequency occurs when setting both registers to their minimum possible values ($\text{PSC} = 0$ and $\text{ARR} = 1$), resulting in $f_{\text{PWM, max}} = \frac{84,000,000}{(0 + 1) \cdot (1 + 1)} = 42\text{ MHz}$; however, because $\text{ARR} = 1$ limits the duty cycle resolution to only two steps ($0\%$ or $100\%$), practical applications requiring fine duty cycle control (e.g., $1\%$ steps) typically cap the maximum operational frequency at $840\text{ kHz}$.

```c
/*----------------------------------------------------------------\
@ Embedded Controller by Young-Keun Kim - Handong Global University
Author           : Choi Junghyun
Created          : 11-07-2026
Modified         : 11-07-2026
Language/ver     : C / C++ in VS Code

#include "stm32f4xx.h"
#include "ecRCC2.h"
#include "ecGPIO2.h"
#include "ecPWM2.h"
#include "ecTIM2.h"
#include "ecEXTI2.h"
#include "ecUART2.h"  
#include <stdio.h>

#define PWM_PIN       PA_0       // TIM2_CH1
#define DIR_PIN       PC_2       // motor direction
#define BUTTON_PIN    PC_13      // speed button B1
#define LED_PIN       PA_5       // Onboard LED
#define IR_PIN        PB_0       // IR sensor

#define IR_DETECTED_LEVEL  0     // LOW(0) = Object detected (Face nearby)

typedef enum {
    FAN_OFF,
    FAN_MID,
    FAN_HIGH,
    FAN_PAUSE
} FanState_t;

volatile FanState_t fanState = FAN_OFF;
volatile FanState_t previousState = FAN_OFF; // Stores active mode before stopping

volatile uint8_t button_event = 0;
volatile uint8_t button_lockout = 0;

volatile uint8_t blink_event = 0;
volatile uint8_t serial_event = 0;

volatile int blink_counter = 0;
volatile int serial_counter = 0;

uint8_t led_state = 0;
float currentDuty = 0.0f;

// retarget standard C stdout printf to UART2 for Terra Term for display
int __io_putchar(int ch) {
    USART2_write((uint8_t*)&ch, 1);
    return ch;
}


void setup(void);
void applyOutputs(void);
void changeMode(void);
const char* getStateName(FanState_t state);

void TIM3_IRQHandler(void);
void EXTI15_10_IRQHandler(void);


int main(void) {
    setup();

    while (1) {
        // check IR sensor
        int objectDetected = (GPIO_read(IR_PIN) == IR_DETECTED_LEVEL);

        // Object detected -> Transition to PAUSE
        if (objectDetected && fanState != FAN_PAUSE) {
            previousState = fanState;       // Save active running state
            fanState = FAN_PAUSE;
            
            button_event = 0;               // Clear pending button events
            led_state = 0;
            GPIO_write(LED_PIN, 0);

            applyOutputs();
        }
        // Object away -> RESUME previous mode
        else if (!objectDetected && fanState == FAN_PAUSE) {
            fanState = previousState;       // Restore saved state
            led_state = 0;

            applyOutputs();
        }

        // Button event
        if (button_event) {
            button_event = 0;

            // Button B1 MUST NOT change mode during PAUSE
            if (fanState != FAN_PAUSE) {
                changeMode();
                applyOutputs();
            }
        }

        
        // LED blinking
        if (blink_event) {
            blink_event = 0;

            if (fanState == FAN_PAUSE) {
                led_state = !led_state;
                GPIO_write(LED_PIN, led_state);
            }
        }

        // terra form displayu
        static FanState_t lastPrintedState = (FanState_t)-1;
        
        if (serial_event) {
            serial_event = 0;

            if (fanState != FAN_OFF || fanState != lastPrintedState) {
                printf("State: %-5s | PWM Duty: %3.0f%% | IR Sensor: %s\r\n",
                   getStateName(fanState),
                   currentDuty * 100.0f,
                   (fanState == FAN_PAUSE) ? "DETECTED (PAUSED)" : "CLEAR");

                lastPrintedState = fanState; // Update last printed state
            }
        }
    }
}


void setup(void) {
    // System Clock
    RCC_PLL_init();

    // UART Initialization for Terra Term (9600 Baud)
    USART2_init();

    // GPIO Configuration
    GPIO_init(DIR_PIN, OUTPUT);
    GPIO_write(DIR_PIN, 0);// Set forward direction

    GPIO_init(LED_PIN, OUTPUT);
    GPIO_write(LED_PIN, 0);// Initial LED state OFF

    GPIO_init(IR_PIN,0);
    GPIO_pupd(IR_PIN, EC_PU);              
    // Pull-up for IR proximity digital pin

    GPIO_init(BUTTON_PIN, 0);
    GPIO_pupd(BUTTON_PIN, EC_PU);          
    // Pull-up for User Button PC13

    // PWM Configuration (1kHz Frequency on PA0)
    PWM_init(PWM_PIN, 1);// 1ms period = 1kHz
    PWM_duty(PWM_PIN, 0.0f);// Initial duty 0%

    //Timer Interrupt Setup(TIM3 - 500ms Update Period)
    TIM_UI_init(TIM3, 500);
    TIM_UI_enable(TIM3);
    NVIC_SetPriority(TIM3_IRQn, 1);
    NVIC_EnableIRQ(TIM3_IRQn);

    //External Interrupt Setup for PC13 Button
    EXTI_init(BUTTON_PIN, FALL, 0);
    NVIC_SetPriority(EXTI15_10_IRQn, 0);
    NVIC_EnableIRQ(EXTI15_10_IRQn);
}
// changing fan speed from 0-> mid -> max
void changeMode(void) {
    switch (fanState) {
        case FAN_OFF:
            fanState = FAN_MID;
            break;
        case FAN_MID:
            fanState = FAN_HIGH;
            break;
        case FAN_HIGH:
            fanState = FAN_OFF;
            break;
        default:
            break;
    }
}

void applyOutputs(void) {
    switch (fanState) {
        case FAN_OFF:
            currentDuty = 0.0f;
            PWM_duty(PWM_PIN, currentDuty);
            GPIO_write(LED_PIN, 0);
            break;

        case FAN_MID:
            currentDuty = 0.5f;
            PWM_duty(PWM_PIN, currentDuty);
            GPIO_write(LED_PIN, 1);
            break;

        case FAN_HIGH:
            currentDuty = 1.0f;
            PWM_duty(PWM_PIN, currentDuty);
            GPIO_write(LED_PIN, 1);
            break;

        case FAN_PAUSE:
            currentDuty = 0.0f;
            PWM_duty(PWM_PIN, currentDuty);
            // LED blinking is handled dynamically in main loop via blink_event
            break;
    }
}


// get state names for display

const char* getStateName(FanState_t state) {
    switch (state) {
        case FAN_OFF:   return "OFF";
        case FAN_MID:   return "MID";
        case FAN_HIGH:  return "HIGH";
        case FAN_PAUSE: return "PAUSE";
        default:        return "UNKNOWN";
    }
}

// TIM3 interrupt
void TIM3_IRQHandler(void) {
    if (is_UIF(TIM3)) {
        clear_UIF(TIM3);

        // Button debounce lockout counter
        if (button_lockout > 0) {
            button_lockout--;
        }

        // LED toggle flag every 1 second(2*500ms)
        blink_counter++;
        if (blink_counter >= 2) {
            blink_counter = 0;
            blink_event = 1;
        }

        //Serial print flag every 2 seconds (4*500ms)
        serial_counter++;
        if (serial_counter >= 4) {
            serial_counter = 0;
            serial_event = 1;
        }
    }
}

// EXTI interrupt handler
void EXTI15_10_IRQHandler(void) {
    if (is_pending_EXTI(BUTTON_PIN)) {
        clear_pending_EXTI(BUTTON_PIN);

        // Debounce lock: Ignore additional edges within lockout window
        if (button_lockout == 0) {
            button_lockout = 1; // Locked out for 500ms

            // Ignore button presses during PAUSE state
            if (fanState != FAN_PAUSE) {
                button_event = 1;
            }
        }
    }
}
```
